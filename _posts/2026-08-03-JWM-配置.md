### 组件与安装

| 组件 | 来源 | 说明 |
|---|---|---|
| JWM | 系统包 | `~/.jwmrc` + `~/.jwm/menu.xml` |
| gen-jwm-menu.py | 自写脚本 | `~/bin/gen-jwm-menu.py` 生成菜单 |

```bash
apt install jwm
```

启动入口见 [dwm 配置与美化](2026-08-03-dwm-配置与美化.md) 中的 `~/.xinitrc`——当前默认 `exec jwm`，autostart 由 `.xinitrc` 自动调用。

### ~/.jwm/autostart.sh

由 `.xinitrc` 调用：

```sh
#!/bin/sh
nm-applet &
pasystray &
```

### gen-jwm-menu.py（菜单生成器）

**Python 3，实际使用**：从 .desktop 文件生成 `~/.jwm/menu.xml`。运行 `python3 ~/bin/gen-jwm-menu.py`：

```python
#!/usr/bin/env python3
"""Generate JWM menu XML from .desktop files."""
import os, subprocess, sys
from configparser import ConfigParser
from pathlib import Path

XDG_DATA_DIRS = os.environ.get("XDG_DATA_DIRS", "/usr/share:/usr/local/share")
APP_DIRS = [f"{d}/applications" for d in XDG_DATA_DIRS.split(":")]
OUTPUT = os.path.expanduser("~/.jwm/menu.xml")
CAT_ORDER = [
    "Network", "Development", "Graphics", "AudioVideo", "Office",
    "Game", "Settings", "System", "Utility",
]
CAT_LABELS = {
    "Network": "Network", "Development": "Development", "Graphics": "Graphics",
    "AudioVideo": "Multimedia", "Office": "Office", "Game": "Games",
    "Settings": "Settings", "System": "System", "Utility": "Accessories",
}

def parse_desktop(path):
    cfg = ConfigParser(strict=False, interpolation=None)
    try:
        cfg.read(path)
    except Exception:
        return None
    if not cfg.has_section("Desktop Entry"):
        return None
    entry = dict(cfg.items("Desktop Entry"))
    if entry.get("no_display", "false").lower() == "true":
        return None
    if entry.get("hidden", "false").lower() == "true":
        return None
    if "try_exec" in entry and not shutil.which(entry["try_exec"].split()[0]):
        return None
    name = entry.get("name") or Path(path).stem
    exec_cmd = entry.get("exec", "")
    icon = entry.get("icon", "")
    cats = [c.strip() for c in entry.get("categories", "").split(";") if c.strip()]
    return {"name": name, "exec": exec_cmd, "icon": icon, "categories": cats}

def main():
    apps = []
    for d in APP_DIRS:
        p = Path(d)
        if not p.is_dir():
            continue
        for f in p.glob("*.desktop"):
            app = parse_desktop(f)
            if app:
                apps.append(app)

    # Group by category
    grouped = {}
    for app in apps:
        assigned = False
        for cat in CAT_ORDER:
            if any(c.startswith(cat) for c in app["categories"]):
                grouped.setdefault(cat, []).append(app)
                assigned = True
                break
        if not assigned:
            grouped.setdefault("Utility", []).append(app)

    # Sort apps by name within each group
    for cat in grouped:
        grouped[cat].sort(key=lambda a: a["name"].lower())

    # Write XML
    lines = ['<?xml version="1.0" encoding="utf-8"?>', '<JWM>']
    for cat in CAT_ORDER:
        if cat not in grouped:
            continue
        label = CAT_LABELS.get(cat, cat)
        lines.append(f'  <Menu label="{label}">')
        for app in grouped[cat]:
            name = app["name"].replace("&", "&amp;").replace('"', "&quot;")
            exec_cmd = app["exec"].replace("&", "&amp;").replace('"', "&quot;")
            if app["icon"]:
                icon = app["icon"].replace("&", "&amp;")
                lines.append(f'    <Program label="{name}" icon="{icon}">{exec_cmd}</Program>')
            else:
                lines.append(f'    <Program label="{name}">{exec_cmd}</Program>')
        lines.append("  </Menu>")
    lines.append("</JWM>")

    os.makedirs(os.path.dirname(OUTPUT), exist_ok=True)
    with open(OUTPUT, "w") as f:
        f.write("\n".join(lines) + "\n")

    print(f"Generated {OUTPUT} with {len(apps)} apps across {len(grouped)} categories")
    return 0

if __name__ == "__main__":
    import shutil
    sys.exit(main())
```

> 注：另有旧版 `~/.local/share/bin/jwm-menu-gen.py`（bashrc alias `jmenu` 指向，中文分类，输出 `~/.config/jwm/generated-menu.xml`）和 `~/.local/bin/jwm-menu-gen`（bash 版），当前均未用于生成 `~/.jwm/menu.xml`。
